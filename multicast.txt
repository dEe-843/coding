bool validateRadioID(const char *id)
{
    if (id == NULL || id[0] == '\0')
        return false;

    for (int i = 0; id[i] != '\0'; i++)
    {
        if (id[i] < '0' || id[i] > '9')
            return false;
    }

    return true;
}


//==================================================
// ETHERNET IP VALIDATION
//==================================================

bool validateEthernetIP(const char *ip)
{
    if (ip == NULL || ip[0] == '\0')
        return false;

    int octetCount = 0;
    int value = 0;
    int digits = 0;

    for (int i = 0; ; i++)
    {
        char ch = ip[i];

        if (ch >= '0' && ch <= '9')
        {
            value = value * 10 + (ch - '0');
            digits++;

            if (digits > 3 || value > 255)
                return false;
        }
        else if (ch == '.' || ch == '\0')
        {
            if (digits == 0)
                return false;

            octetCount++;

            if (octetCount > 4)
                return false;

            value = 0;
            digits = 0;

            if (ch == '\0')
                break;
        }
        else
        {
            return false;
        }
    }

    return (octetCount == 4);
}


//==================================================
// APPEND RADIO ID
//==================================================

bool appendRadioID(const char *id)
{
    if (radioIDCount >= MAX_DESTINATIONS)
        return false;

    if (!validateRadioID(id))
        return false;

    if (strlen(IPAddr_line) + strlen(id) + 2 >=
        sizeof(IPAddr_line))
        return false;

    strcat(IPAddr_line, id);

    strcat(IPAddr_line, "|");

    radioIDCount++;

    return true;
}


//==================================================
// APPEND ETHERNET IP
//==================================================

bool appendEthernetIP(const char *ip)
{
    if (ethernetIPCount >= MAX_DESTINATIONS)
        return false;

    if (!validateEthernetIP(ip))
        return false;

        sizeof(EthernetAddr_line))
        return false;

    strcat(EthernetAddr_line, ip);

    strcat(EthernetAddr_line, "|");

    ethernetIPCount++;

    return true;
}

void extractDestinations(const char *buffer)
{
    char temp[BUFFER_SIZE];

    destinationCount = 0;

    strncpy(temp, buffer, sizeof(temp) - 1);

    temp[sizeof(temp) - 1] = '\0';

    char *token = strtok(temp, "|");

    while (token != NULL &&
           destinationCount < MAX_DESTINATIONS)
    {
        strncpy(sendStatus[destinationCount].destination,
                token, INPUT_SIZE - 1);

        sendStatus[destinationCount].destination[
            INPUT_SIZE - 1] = '\0';

        sendStatus[destinationCount].sent = false;

        sendStatus[destinationCount].delivered = false;

        sendStatus[destinationCount].retryCount = 0;

        destinationCount++;

        token = strtok(NULL, "|");
    }
}

//==================================================
// ACK RECEIVED CALLBACK
//==================================================

// Call this only after verifying that the received ACK
// belongs to the current destination.

void destinationACKReceived(const char *ackDestination)
{
    if (!waitingForACK)
        return;

    if (ackDestination == NULL)
        return;

    if (strcmp(
        ackDestination,
        sendStatus[currentDestination].destination) != 0)
    {
        return;
    }

    sendStatus[currentDestination].delivered = true;

    waitingForACK = false;

    updateStatusTable();

    currentDestination++;

    sendNextDestination();
}


//==================================================
// ACK TIMEOUT CALLBACK
//==================================================

void destinationACKTimeout(void)
{
    if (!waitingForACK)
        return;

    if (sendStatus[currentDestination].retryCount <
        MAX_RETRIES)
    {
        sendStatus[currentDestination].retryCount++;

        // Retry the same destination.

        sendNextDestination();

        return;
    }

    // Maximum retries reached.

    sendStatus[currentDestination].delivered = false;

    waitingForACK = false;

    updateStatusTable();

    currentDestination++;

    sendNextDestination();
}

# 2. How the complete flow works

Radio interface

Radio button → Manual ID / Multicast

Validate ID → Append to `IPAddr_line`

Send button → Extract `|`

Send individual Radio IDs → Wait for ACK

Update Sent and Delivered columns

Ethernet interface

Ethernet button → Manual IP / Multicast

Validate IP → Append to `EthernetAddr_line`

Send button → Extract `|`

Separate IP into four octets

Send individual IPs → Wait for ACK

Update Sent and Delivered columns

# 3. Important integration points

There are four places where you must connect your existing MCC/Legato implementation:

| Function                   | What you must connect                      |
| -------------------------- | ------------------------------------------ |
| `manualAddButtonClicked()` | Read text from your actual text field      |
| `updateStatusTable()`      | Update your actual Label widgets           |
| `sendRadioPacket()`        | Your existing Radio packet transmission    |
| `sendEthernetPacket()`     | Your existing Ethernet packet transmission |

Also, `destinationACKTimeout()` must be called by your PIC timer callback. The ACK retry function should restart the timer for every retry.

Important correction for ACK retries: In the retry section, call the packet transmission function again for the current destination and restart the timer. Do not call `sendNextDestination()` directly as written in the initial retry skeleton, because that function advances only after transmission failure and is not a dedicated retry routine.
